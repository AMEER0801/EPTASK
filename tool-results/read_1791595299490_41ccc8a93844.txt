    90→class TocDocTemplate(SimpleDocTemplate):
    91→    def afterFlowable(self, flowable):
    92→        if hasattr(flowable, "bookmark_name"):
    93→            level = flowable.bookmark_level
    94→            text = flowable.bookmark_text
    95→            display_page = max(1, self.page - 1)  # body pages display 1-based after the TOC page
    96→            key = flowable.bookmark_key
    97→            self.notify("TOCEntry", (level, text, display_page, key))  # 4-tuple -> clickable links
    98→            self.canv.bookmarkPage(key)
    99→            self.canv.addOutlineEntry(text, key, level=level, closed=False)

   102→def head(text, level=0):
   103→    key = f"h{level}-{abs(hash(text)) % 99999}"
   104→    style = h1 if level == 0 else h2
   105→    p = Paragraph(f'<a name="{key}"/><b>{text}</b>', style)
   106→    p.bookmark_name = key
   107→    p.bookmark_level = level
   108→    p.bookmark_text = text
   109→    p.bookmark_key = key
   110→    return p

   113→def footer(canvas, doc):
   114→    canvas.saveState()
   115→    canvas.setFont("FreeSerif", 9)
   116→    canvas.setFillColor(TEXT_MUTED)
   117→    if doc.page == 1:
   118→        label = "i"                      # front matter (contents page)
   119→    else:
   120→        label = str(doc.page - 1)        # body resets to arabic 1
   121→    canvas.drawCentredString(PAGE_W / 2, 0.5 * inch, label)
   122→    canvas.setFont("FreeSerif-Italic", 8)
   123→    canvas.drawRightString(PAGE_W - MARGIN, 0.5 * inch, "EPtask Ops Console - Integration Report")
   124→    canvas.restoreState()

   130→def safe_keep(elements):
   131→    total = 0
   132→    for el in elements:
   133→        w, hgt = el.wrap(AVAIL_W, PAGE_H)
   134→        total += hgt
   135→    if total <= MAX_KEEP:
   136→        return [KeepTogether(elements)]
   137→    if len(elements) >= 2:
   138→        return [KeepTogether(elements[:2])] + list(elements[2:])
   139→    return list(elements)

   142→def make_table(header, rows, widths, mono_cols=(), stripe_from=1):
   143→    data = [[Paragraph(f"<b>{c}</b>", th) for c in header]]
   144→    for r_i, row in enumerate(rows):
   145→        cells = []
   146→        for c_i, c in enumerate(row):
   147→            st = td_mono if c_i in mono_cols else td_
   148→            cells.append(Paragraph(str(c), st))
   149→        data.append(cells)
   150→    scale = min(1.0, AVAIL_W / sum(widths))
   151→    widths = [w * scale for w in widths]
   152→    t = Table(data, colWidths=widths, hAlign="CENTER", repeatRows=1)
   153→    style = [
   154→        ("BACKGROUND", (0, 0), (-1, 0), HEADER_FILL),
   155→        ("GRID", (0, 0), (-1, -1), 0.5, BORDER),
   156→        ("VALIGN", (0, 0), (-1, -1), "MIDDLE"),
   157→        ("LEFTPADDING", (0, 0), (-1, -1), 6),
   158→        ("RIGHTPADDING", (0, 0), (-1, -1), 6),
   159→        ("TOPPADDING", (0, 0), (-1, -1), 5),
   160→        ("BOTTOMPADDING", (0, 0), (-1, -1), 5),
   161→    ]
   162→    for i in range(stripe_from, len(data)):
   163→        if (i - stripe_from) % 2 == 1:
   164→            style.append(("BACKGROUND", (0, i), (-1, i), TABLE_STRIPE))
   165→    t.setStyle(TableStyle(style))
   166→    return t
